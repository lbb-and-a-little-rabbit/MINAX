if(NOT TARGET minax_compiler)
    message(FATAL_ERROR "Configure from compiler: cmake -S compiler -B build")
endif()
add_library(minax_evaluator SHARED src/Evaluator.cpp)
target_include_directories(minax_evaluator PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(minax_evaluator PUBLIC minax_compiler)
if(WIN32)
    set_target_properties(minax_evaluator PROPERTIES PREFIX "")
endif()
if(MINGW)
    target_link_options(minax_evaluator PRIVATE "-Wl,--export-all-symbols")
endif()
minax_copy_runtime(minax_evaluator)

add_executable(EvaluatorDemo src/main.cpp)
target_link_libraries(EvaluatorDemo PRIVATE minax_evaluator)
minax_copy_runtime(EvaluatorDemo)

if(BUILD_TESTING)
    add_executable(EvaluatorTest test/EvaluatorTest.cpp)
    target_link_libraries(EvaluatorTest PRIVATE minax_evaluator)
    minax_copy_runtime(EvaluatorTest)
    add_test(NAME EvaluatorTest COMMAND $<TARGET_FILE:EvaluatorTest>)
endif()
